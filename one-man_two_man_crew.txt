@name shitbox/one-man two man crew
@inputs EngineerSeat:entity Teleport Enable
@outputs [O ConductorSeat]:entity ConductorPos:table 
@persist [ConductorModel StandAnim AirAnim SitAnim AttachmentName]:string LockPod ShowConductor AnimOverrides:table


if(first()|dupefinished()){

#[

    Thank you for using the One-Man Two Man Crew E2 v0.3
    This E2 can be found in the Tree Pyro's Shitbox Github repo:
    https://github.com/TreePyro/treepyros_shitbox
    
    This E2 works by teleporting you to the Engineer Seat when you use Teleport while not in the seat.
    If you use the E2 while inside the Engineer Seat, it teleports you to the Conductor's position or seat.
    The Conductor's position or seat is recorded when using Teleport outside of EngineerSeat.
    
    A holo model may also be placed where the aforementioned recorded position is.
    
    To set this up: All you need to do is place this e2 literally anywhere on your locomotive.
    ->Wire "EngineerSeat" to the driver seat of your vehicle.
    ->Wire "Teleport" to a Wiremod Numpad Input, with a key set to whatever key you want to press to teleport.
    ->Wire "Enable" to a toggled button that is only activated when you're using the vehicle's controls
    (Note: Do not use "Active", as it will disable your ability to teleport back into the engineer seat)
    
    If you do not intend to use multiple locomotives, you can wire Enable to the same button as Teleport.
    
]#

    #Enable or disable the conductor holo
    #Default 1
    ShowConductor = 1
    
    #The model that the conductor uses
    #Default: models/player/group03/male_07.mdl
    ConductorModel = "models/player/group03/male_07.mdl"
    
    #And you can also set the animations your conductor uses here:
    StandAnim = "idle_all_01" #Default "idle_all_01"
    AirAnim = "swim_idle_all" #Default "swim_idle_all"
    SitAnim = "sit" #Default "sit"
    
    #Set this to 1 if you don't want your conductor's seat to be used whenever he's riding it.
    #MAY BE BUGGY
    LockPod = 0
    
    #You can also use this table to set individual animations for different seat models.
    #If the conductor is in a seat with any of these models, he will use the described animation.
    #Otherwise, he will use SitAnim.
    AnimOverrides = table(
        "models/trakpak3_common/vehicles/footprints.mdl" = StandAnim,
        "models/magtrainslocos/cabseats/footplate.mdl" = StandAnim,
        "models/chairs_playerstart/standingpose.mdl" = StandAnim
    )

    #[ Default values:
    AnimOverrides = table(
        "models/trakpak3_common/vehicles/footprints.mdl" = StandAnim,
        "models/magtrainslocos/cabseats/footplate.mdl" = StandAnim,
        "models/chairs_playerstart/standingpose.mdl" = StandAnim
    )
    ]#
        
    #/// END OF CONFIG ///#
    #You should not change anything below this line unless you know what you're doing.
    
    
    
    
    
    
    
    
    
    
    
    AttachmentName = "vehicle_feet_passenger0"
    O = owner()
    ConductorSeat = _NO_ENTITY
    ConductorPos = table(
        "pos" = O:pos(),
        "ang" = O:angles()
    )
    
    if(ShowConductor){
        holoCreate(0)
        holoModel(0,ConductorModel)
        holoPos(0,vec(0))
    }
}

event keyPressed(Ply:entity, Key:string, Dn:number, _:string){
    if(Ply == O & Teleport & Enable){
        if(O:vehicle() != EngineerSeat){
            if(O:inVehicle()){
                ConductorSeat = O:vehicle()
                if(LockPod){ConductorSeat:lockPod(1)}
                ConductorPos = table(
                    "pos" = ConductorSeat:pos(),
                    "ang" = ConductorSeat:toWorld(ang(0,-90,0)),
                    "noclip" = 0
                )
                
                if(ShowConductor){
                    holoPos(0,ConductorSeat:attachmentPos(AttachmentName))
                    holoParentAttachment(0,ConductorSeat,AttachmentName)
                    holoAng(0,ConductorSeat:toWorld(ang(0,90,0)))

                    let Mdl = ConductorSeat:model()
                    if(AnimOverrides:exists(Mdl)){
                        holoAnim(0,AnimOverrides[Mdl,string])
                    }else{
                        holoAnim(0,SitAnim)
                    }
                }
                
                O:plyExitVehicle()
                O:plyEnterVehicle(EngineerSeat)
            }else{
                ConductorSeat:lockPod(0)
                ConductorSeat = _NO_ENTITY
                ConductorPos = table(
                    "pos" = O:pos(),
                    "ang" = O:angles(),
                    "noclip" = O:inNoclip()
                )
                O:plyEnterVehicle(EngineerSeat)
                
                if(ShowConductor){
                    holoUnparent(0)
                    holoPos(0,ConductorPos["pos",vector])
                    holoAng(0,ConductorPos["ang",angle]:setRoll(0))
                    holoAnim(0,O:isOnGround() ? StandAnim : AirAnim)
                }
            }
        }else{
            if(ShowConductor){holoPos(0,vec(0))}
            O:plyExitVehicle()
            if(ConductorSeat:isValid()){
                ConductorSeat:lockPod(0)
                O:plyEnterVehicle(ConductorSeat)
            }else{
                O:plySetPos(ConductorPos["pos",vector])
                O:plySetAng(ConductorPos["ang",angle])
                O:plyNoclip(ConductorPos["noclip",number])
            }
        }
    }
}

if(~Enable & !Enable){
    holoDelete(0)
}
